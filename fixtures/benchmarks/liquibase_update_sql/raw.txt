-- /* batched retention purge of the *_Run/*_RunDetail audit logs (DemoRefresh, PIIMasking, PIIAnonymization); keeps the newest @KeepRuns runs per family, child-first; OBJECT_ID-guarded so it no-ops where a family's tables are absent */
CREATE OR ALTER PROCEDURE dbo.usp_purge_run_audit
    @KeepRuns     int = 10,
    @BatchSize    int = 5000,
    @PauseSeconds int = 1,
    @DryRun       bit = 0
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    IF @KeepRuns  < 0 SET @KeepRuns  = 0;
    IF @BatchSize < 1 SET @BatchSize = 5000;

    DECLARE @pause    varchar(8) = CONVERT(varchar(8), DATEADD(SECOND, @PauseSeconds, '00:00:00'), 108);
    DECLARE @cutoffId bigint;
    DECLARE @deleted  int;




    IF OBJECT_ID(N'CICD.dbo.DemoRefreshRun', N'U') IS NOT NULL
    BEGIN
        SET @cutoffId = NULL;
        SELECT @cutoffId = MIN(k.Id)
        FROM (SELECT TOP (@KeepRuns) DemoRefreshRunId AS Id
              FROM CICD.dbo.DemoRefreshRun ORDER BY DemoRefreshRunId DESC) k;

        IF @cutoffId IS NOT NULL
        BEGIN
            IF @DryRun = 1
                SELECT N'DemoRefresh' AS Family,
                       (SELECT COUNT(*) FROM CICD.dbo.DemoRefreshRun       WHERE DemoRefreshRunId < @cutoffId) AS RunsToDelete,
                       (SELECT COUNT(*) FROM CICD.dbo.DemoRefreshRunDetail WHERE DemoRefreshRunId < @cutoffId) AS DetailRowsToDelete;
            ELSE
            BEGIN
                WHILE 1 = 1
                BEGIN
                    DELETE TOP (@BatchSize) FROM CICD.dbo.DemoRefreshRunDetail WHERE DemoRefreshRunId < @cutoffId;
                    SET @deleted = @@ROWCOUNT; IF @deleted = 0 BREAK; WAITFOR DELAY @pause;
                END;
                WHILE 1 = 1
                BEGIN
                    DELETE TOP (@BatchSize) FROM CICD.dbo.DemoRefreshRun WHERE DemoRefreshRunId < @cutoffId;
                    SET @deleted = @@ROWCOUNT; IF @deleted = 0 BREAK; WAITFOR DELAY @pause;
                END;
            END
        END
    END;




    IF OBJECT_ID(N'CICD.dbo.PIIMaskingRun', N'U') IS NOT NULL
    BEGIN
        SET @cutoffId = NULL;
        SELECT @cutoffId = MIN(k.Id)
        FROM (SELECT TOP (@KeepRuns) PIIMaskingRunId AS Id
              FROM CICD.dbo.PIIMaskingRun ORDER BY PIIMaskingRunId DESC) k;

        IF @cutoffId IS NOT NULL
        BEGIN
            IF @DryRun = 1
                SELECT N'PIIMasking' AS Family,
                       (SELECT COUNT(*) FROM CICD.dbo.PIIMaskingRun       WHERE PIIMaskingRunId < @cutoffId) AS RunsToDelete,
                       (SELECT COUNT(*) FROM CICD.dbo.PIIMaskingRunDetail WHERE PIIMaskingRunId < @cutoffId) AS DetailRowsToDelete;
            ELSE
            BEGIN
                WHILE 1 = 1
                BEGIN
                    DELETE TOP (@BatchSize) FROM CICD.dbo.PIIMaskingRunDetail WHERE PIIMaskingRunId < @cutoffId;
                    SET @deleted = @@ROWCOUNT; IF @deleted = 0 BREAK; WAITFOR DELAY @pause;
                END;
                WHILE 1 = 1
                BEGIN
                    DELETE TOP (@BatchSize) FROM CICD.dbo.PIIMaskingRun WHERE PIIMaskingRunId < @cutoffId;
                    SET @deleted = @@ROWCOUNT; IF @deleted = 0 BREAK; WAITFOR DELAY @pause;
                END;
            END
        END
    END;




    IF OBJECT_ID(N'CICD.dbo.PIIAnonymizationRun', N'U') IS NOT NULL
    BEGIN
        SET @cutoffId = NULL;
        SELECT @cutoffId = MIN(k.Id)
        FROM (SELECT TOP (@KeepRuns) PIIAnonymizationRunId AS Id
              FROM CICD.dbo.PIIAnonymizationRun ORDER BY PIIAnonymizationRunId DESC) k;

        IF @cutoffId IS NOT NULL
        BEGIN
            IF @DryRun = 1
                SELECT N'PIIAnonymization' AS Family,
                       (SELECT COUNT(*) FROM CICD.dbo.PIIAnonymizationRun       WHERE PIIAnonymizationRunId < @cutoffId) AS RunsToDelete,
                       (SELECT COUNT(*) FROM CICD.dbo.PIIAnonymizationRunDetail WHERE PIIAnonymizationRunId < @cutoffId) AS DetailRowsToDelete;
            ELSE
            BEGIN
                WHILE 1 = 1
                BEGIN
                    DELETE TOP (@BatchSize) FROM CICD.dbo.PIIAnonymizationRunDetail WHERE PIIAnonymizationRunId < @cutoffId;
                    SET @deleted = @@ROWCOUNT; IF @deleted = 0 BREAK; WAITFOR DELAY @pause;
                END;
                WHILE 1 = 1
                BEGIN
                    DELETE TOP (@BatchSize) FROM CICD.dbo.PIIAnonymizationRun WHERE PIIAnonymizationRunId < @cutoffId;
                    SET @deleted = @@ROWCOUNT; IF @deleted = 0 BREAK; WAITFOR DELAY @pause;
                END;
            END
        END
    END;
END;
GO
